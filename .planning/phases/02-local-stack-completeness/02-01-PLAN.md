import { Context } from '@aws-appsync/utils';

export function wrapResolver(handler: Function) {
  return async (args: any, context: any, info: any) => {
    // Builds appsyncContext: Context from args, context, info
    // context.identity is set directly from Redis CognitoIdentity shape
    // KNOWN GAP: identity.claims is NOT set — will be fixed in Plan 02
    const appsyncContext: Context = {
      arguments: args,
      source: {},
      result: {},
      prev: { result: {} },
      stash: { permissions: context.permissions || {} },
      info: {
        fieldName: info?.fieldName || '',
        parentTypeName: info?.parentType?.name || '',
        variables: info?.variableValues || {},
        selectionSetList: [...],
        selectionSetGraphQL: ''
      },
      request: { headers: context.headers || {}, domainName: null },
      identity: context.identity || null,  // CognitoIdentity shape from Redis
      error: null
    };
    const result = await handler(appsyncContext);
    return result;
  };
}

export function createContext(docClient: any, rabbitChannel: any, identity?: any) {
  return { docClient, rabbitChannel, identity, headers: {} };
}
